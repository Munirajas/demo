Server-Side Amenity Fetchers

Same pattern as server-location-functions.ts: server-only, cache(), typed response, RateLimitError handling

the one unavoidable waterfall

General info must resolve before you know the amenity's type — so "fetch schedule only if Scheduled Admission" can't be parallelized with the general-info fetch itself. Everything else that doesn't depend on that decision (location functions, amenity types dropdown options) still runs alongside it via Promise.all.

2. Server fetchers — mirrors server-location-functions.ts exactly

src/modules/amenities/actions/server-amenities.ts 
import 'server-only';

import { cache } from 'react';
import { serverGqlFetch } from '@/lib/graphql/server-gql-fetcher';
import { RateLimitError } from '@/lib/graphql/gql-errors';
import type {
  AmenityGeneralInfo,
  RawAmenityGeneralInfo,
  AmenitySchedule,
  RawAmenitySchedule,
} from '@/types/amenities';
import { mapAmenityGeneralInfo, mapAmenitySchedule } from '../lib/utils';
import { SCHEDULED_ADMISSION_TYPE_NAME } from '../lib/constants';
import {
  GET_AMENITY_GENERAL_INFO,
  GET_AMENITY_SCHEDULE,
} from '../graphql/amenities.queries';

interface GetAmenityGeneralInfoResponse {
  getAmenityGeneralInfo: {
    success: boolean;
    message: string;
    data: RawAmenityGeneralInfo;
  };
}

export const getAmenityGeneralInfo = cache(
  async (amenityId: number): Promise<AmenityGeneralInfo | null> => {
    try {
      const response = await serverGqlFetch<GetAmenityGeneralInfoResponse>(
        GET_AMENITY_GENERAL_INFO,
        { type: 'client', amenityId },
      );
      if (!response.getAmenityGeneralInfo?.success) {
        return null;
      }
      return mapAmenityGeneralInfo(response.getAmenityGeneralInfo.data);
    } catch (error) {
      if (error instanceof RateLimitError) {
        console.warn('[getAmenityGeneralInfo] Rate limited — returning null');
        return null;
      }
      console.error('[getAmenityGeneralInfo]', error);
      return null;
    }
  },
);

interface GetAmenityScheduleResponse {
  getAmenitySchedule: {
    success: boolean;
    message: string;
    data: RawAmenitySchedule;
  };
}

export const getAmenitySchedule = cache(
  async (amenityId: number): Promise<AmenitySchedule | null> => {
    try {
      const response = await serverGqlFetch<GetAmenityScheduleResponse>(
        GET_AMENITY_SCHEDULE,
        { type: 'client', amenityId },
      );
      if (!response.getAmenitySchedule?.success) {
        return null;
      }
      return mapAmenitySchedule(response.getAmenitySchedule.data);
    } catch (error) {
      if (error instanceof RateLimitError) {
        console.warn('[getAmenitySchedule] Rate limited — returning null');
        return null;
      }
      console.error('[getAmenitySchedule]', error);
      return null;
    }
  },
);

/**
 * Orchestrator for the view/edit page: always fetches general info,
 * then conditionally fetches type-specific detail based on
 * amenityTypeName. Add a new branch here (seatedInfo, pricing, ...)
 * whenever a new amenity type needs its own detail call — the page
 * component itself never has to change.
 */
export async function getAmenityDetails(amenityId: number) {
  const generalInfo = await getAmenityGeneralInfo(amenityId);
  if (!generalInfo) {
    return { generalInfo: null, schedule: null };
  }

  const schedule =
    generalInfo.amenityTypeName === SCHEDULED_ADMISSION_TYPE_NAME
      ? await getAmenitySchedule(amenityId)
      : null;

  // Future: seatedInfo / pricing follow the same conditional pattern here.

  return { generalInfo, schedule };
}

3. View-amenity page.tsx — parallel where possible

src/app/(protected)/location/[slug]/[fn]/(amenity)/view-amenity/[amenityId]/page.tsx 
import { notFound } from 'next/navigation';
import { getLocationContext } from '@/modules/locations/lib/get-location-context';
import { getLocationFunctions } from '@/modules/locations/actions/server-location-functions';
import { getAmenityTypes, getAmenityDetails } from '@/modules/amenities/actions/server-amenities';
import { PROVIDE_AMENITIES_SLUG } from '@/modules/amenities/lib/constants';
import { EditAmenity } from '@/modules/amenities/components/AddAmenity/EditAmenity';

interface Props {
  params: Promise<{ slug: string; fn: string; amenityId: number }>;
}

export default async function ViewAmenityPage({ params }: Props) {
  const { slug, fn, amenityId } = await params;
  if (fn !== 'provide-amenities') notFound();

  const found = await getLocationContext(slug);
  if (!found) notFound();
  const locationId = Number(found.node.id);

  // getAmenityDetails resolves general info first, then conditionally
  // fetches schedule — that pair can't be parallelized. Everything else
  // (location functions, amenity type options) has no such dependency
  // and runs alongside it instead of after it.
  const [{ generalInfo, schedule }, functions, amenityTypesResult] =
    await Promise.all([
      getAmenityDetails(amenityId),
      getLocationFunctions(String(locationId)),
      getAmenityTypes(),
    ]);

  if (!generalInfo) notFound();

  const amenitiesFn = functions.find((f) => f.slug === PROVIDE_AMENITIES_SLUG);

  return (
    <EditAmenity
      amenityId={amenityId}
      locationId={locationId}
      locationSlug={slug}
      amenityTypes={amenityTypesResult.amenityTypes}
      locationFunctionId={amenitiesFn?.locationFunctionId ?? 0}
      generalInfo={generalInfo}
      schedule={schedule}
    />
  );
}

4. EditAmenity.tsx (client) — picks the right form, wires the mutation hooks

src/modules/amenities/components/AddAmenity/EditAmenity.tsx 
'use client';

import { GeneralAdmissionForm } from '../forms/GeneralAdmissionForm';
import { ScheduledAdmissionForm } from '../forms/ScheduledAdmissionForm';
import { useUpdateAmenity } from '@/modules/amenities/hooks/useUpdateAmenity';
import { useUpsertAmenitySchedule } from '@/modules/amenities/hooks/useUpsertAmenitySchedule';
import { SCHEDULED_ADMISSION_TYPE_NAME } from '@/modules/amenities/lib/constants';
import type { AmenityGeneralInfo, AmenitySchedule, AmenityType } from '@/types/amenities';

interface Props {
  amenityId: number;
  locationId: number;
  locationSlug: string;
  amenityTypes: AmenityType[];
  locationFunctionId: number;
  generalInfo: AmenityGeneralInfo;
  schedule: AmenitySchedule | null;
}

export function EditAmenity({
  amenityId,
  locationId,
  locationSlug,
  amenityTypes,
  locationFunctionId,
  generalInfo,
  schedule,
}: Props) {
  const updateAmenity = useUpdateAmenity(amenityId);
  const upsertSchedule = useUpsertAmenitySchedule(amenityId);

  const isScheduled = generalInfo.amenityTypeName === SCHEDULED_ADMISSION_TYPE_NAME;

  return (
    <div className="flex flex-col gap-6">
      <GeneralAdmissionForm
        amenityId={amenityId}
        locationId={locationId}
        amenityTypes={amenityTypes}
        initialValues={generalInfo}
        onSubmit={(values) => updateAmenity.mutate(values)}
        isSubmitting={updateAmenity.isPending}
      />

      {isScheduled && (
        <ScheduledAdmissionForm
          amenityId={amenityId}
          locationFunctionId={locationFunctionId}
          initialSchedule={schedule}
          onSubmit={(values) => upsertSchedule.mutate(values)}
          isSubmitting={upsertSchedule.isPending}
        />
      )}
    </div>
  );
}

Both forms receive server-fetched data as initialValues/initialSchedule — no client-side fetch or loading spinner on mount. Submission (edit) stays entirely client-side through your existing useUpdateAmenity / useUpsertAmenitySchedule mutation hooks, which already handle cache invalidation/refetch after a successful save.
