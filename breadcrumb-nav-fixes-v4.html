Breadcrumb & Active-Nav Fixes

Fixes: dead "provide-amenities" crumb, raw-slug label, inactive left nav on nested routes

root cause

Bug 1 & 2 (dead link, raw-slug label): the previous baseHref calculation used string-suffix slicing on pathname, which is fragile. It silently produced an unusable href, so the crumb fell back to the non-link, unmapped-label render path.

Bug 3 (nav not active): FunctionsNav uses pathname === href (exact match) — a nested URL like .../provide-amenities/view-amenity/184 never equals .../provide-amenities, so it never lights up.

1. Breadcrumbs.tsx — robust href + label resolution

src/components/layout/Breadcrumbs.tsx 
'use client';

import { useMemo, useState } from 'react';
import Link from 'next/link';
import { usePathname, useSelectedLayoutSegments } from 'next/navigation';
import type { BreadcrumbItem } from '@/types/locations';
import { ROUTE_SEGMENT_LABELS } from '@/lib/breadcrumb-segments';

const MAX_LOCATION_CRUMBS = 3;

interface Props {
  items: BreadcrumbItem[];
}

export function Breadcrumbs({ items }: Props) {
  const [open, setOpen] = useState(false);
  const pathname = usePathname();
  const nestedSegments = useSelectedLayoutSegments();

  const allItems = useMemo(
    () => appendNestedCrumbs(items, nestedSegments, pathname),
    [items, nestedSegments, pathname],
  );

  if (!allItems.length) return null;

  const fnItem = allItems[allItems.length - 1];
  const locationItems = allItems.slice(0, -1);
  const needsCollapse = locationItems.length > MAX_LOCATION_CRUMBS;
  const visibleLocations = needsCollapse
    ? [locationItems[0], ...locationItems.slice(-(MAX_LOCATION_CRUMBS - 1))]
    : locationItems;
  const hiddenLocations = needsCollapse
    ? locationItems.slice(1, -(MAX_LOCATION_CRUMBS - 1))
    : [];

  return (
    <nav className="flex items-center gap-1 flex-wrap text-sm min-w-0">
      {visibleLocations.map((item, i) => (
        <span key={item.href ?? item.label} className="flex items-center gap-1 min-w-0">
          {i > 0 && <Sep />}
          <Crumb item={item} />
          {needsCollapse && i === 0 && (
            <>
              <Sep />
              <div className="relative">
                <button type="button" onClick={() => setOpen((o) => !o)}
                  className="px-2 py-0.5 text-sm text-gray-500 hover:text-gray-900 hover:bg-gray-50 rounded-md">
                  …
                </button>
                {open && (
                  <>
                    <div className="fixed inset-0 z-40" onClick={() => setOpen(false)} />
                    <div className="absolute left-0 top-full mt-1 min-w-[180px] bg-white rounded-lg shadow-lg border border-gray-100 z-50 overflow-hidden py-1">
                      {hiddenLocations.map((h) => (
                        <Link key={h.href} href={h.href ?? '#'} onClick={() => setOpen(false)}
                          className="block px-3 py-2 text-sm text-gray-600 hover:bg-gray-50 hover:text-gray-900 transition-colors truncate">
                          {h.label}
                        </Link>
                      ))}
                    </div>
                  </>
                )}
              </div>
            </>
          )}
        </span>
      ))}
      <Sep />
      <span className="px-4 py-0.5 font-medium text-blue-700 bg-blue-50 rounded-md" title={fnItem.label}>
        {fnItem.label}
      </span>
    </nav>
  );
}

/**
 * Appends crumbs for route segments below [fn], and resolves display
 * labels for BOTH the fn crumb and nested segments from one shared map —
 * this is what fixes the raw "provide-amenities" text bug, since the
 * label no longer depends on whatever buildBreadcrumbs happened to pass in.
 *
 * href for the fn crumb is computed by removing exactly `segments.length`
 * parts from the END of the pathname's path segments (array-based, not
 * string-suffix slicing) — this is what fixes the dead-link bug, since it
 * can't be thrown off by trailing slashes or encoding quirks.
 */
function appendNestedCrumbs(
  baseItems: BreadcrumbItem[],
  segments: string[],
  pathname: string,
): BreadcrumbItem[] {
  if (!baseItems.length) return baseItems;

  const pathParts = pathname.split('/').filter(Boolean);
  const baseParts = segments.length
    ? pathParts.slice(0, pathParts.length - segments.length)
    : pathParts;
  const baseHref = '/' + baseParts.join('/');
  const fnSlug = baseParts[baseParts.length - 1];

  const items = [...baseItems];
  const last = items[items.length - 1];
  items[items.length - 1] = {
    ...last,
    label: ROUTE_SEGMENT_LABELS[fnSlug] ?? last.label,
    href: baseHref,
  };

  if (!segments.length) return items;

  const extra: BreadcrumbItem[] = [];
  let hrefSoFar = baseHref;

  for (const seg of segments) {
    hrefSoFar = `${hrefSoFar}/${seg}`;
    const label = ROUTE_SEGMENT_LABELS[seg];
    if (label) {
      extra.push({ label, href: hrefSoFar });
    } else if (extra.length) {
      // dynamic id (e.g. amenityId) folds into the previous crumb —
      // "view-amenity/184" stays ONE link, ending in /184
      extra[extra.length - 1] = { ...extra[extra.length - 1], href: hrefSoFar };
    }
  }

  return [...items, ...extra];
}

function Sep() {
  return <span className="text-gray-300 select-none">/</span>;
}

function Crumb({ item }: { item: BreadcrumbItem }) {
  if (item.href) {
    return (
      <Link href={item.href}
        className="px-4 py-0.5 text-sm text-left rounded-md transition-colors text-gray-900 hover:text-blue-700 hover:bg-blue-50 font-medium border-transparent cursor-pointer"
        title={item.label}>
        {item.label}
      </Link>
    );
  }
  return <span className="px-4 py-0.5 text-sm text-gray-500" title={item.label}>{item.label}</span>;
}

2. Shared label map (single source of truth)

src/lib/breadcrumb-segments.ts 
/**
 * Display label for ANY route segment — top-level fn or nested —
 * keyed by the raw URL slug. One map, one place to extend.
 */
export const ROUTE_SEGMENT_LABELS: Record<string, string> = {
  // top-level functions
  'provide-amenities': 'Provide Amenities',
  'plan-selection': 'Plan Selection',
  configurations: 'Configurations',
  tickets: 'Tickets',

  // nested amenity routes
  'add-amenity': 'Add Amenity',
  'view-amenity': 'View Amenity',
};

Reuse this same map for FunctionsNav's display labels too if it currently uses raw fn.functionTypeName text — keeps every label consistent across nav, breadcrumb, and page titles from one file.
