2. Shared selectors — the ONE place this mapping logic lives

src/modules/amenities/lib/selectors.ts 
import type {
  GeneralInfoValues,
  GetAmenityGeneralInfoResponse,
  GetAmenityScheduleDetailsResponse,
  AmenitySchedule,
} from '@/types/amenities';

/**
 * Same logic that was inline in useAmenityGeneralInfo's `select` — moved
 * here so the server fetcher can call the exact same function instead of
 * re-implementing the mapping.
 */
export function selectGeneralInfo(
  data: GetAmenityGeneralInfoResponse,
): GeneralInfoValues | null {
  const d = data?.getAmenityDetail?.data;
  if (!d?.[0]) return null;

  const { amenityTypeId, amenityName, amenityDescription } = d[0];
  return {
    amenityTypeId,
    amenityName,
    amenityDescription: amenityDescription ?? '',
  };
}

/** Same logic that was inline in useAmenitySchedule's `select`. */
export function selectSchedule(
  data: GetAmenityScheduleDetailsResponse,
): AmenitySchedule | null {
  return data.getAmenitySchedule?.data ?? null;
}

3. Client hooks — now call the shared selector instead of inlining it

src/modules/amenities/hooks/useAmenityGeneralInfo.ts 
'use client';

import { useAuthQuery } from '@/modules/auth/hooks/useAuthQuery';
import { amenityKeys, AMENITY_REQUEST_TYPE } from '../lib/constants';
import { selectGeneralInfo } from '../lib/selectors';
import { GET_AMENITY_DETAIL } from '../graphql/amenity.queries';

export function useAmenityGeneralInfo(amenityId: number | null) {
  return useAuthQuery(
    amenityKeys.generalInfo(amenityId ?? -1),
    GET_AMENITY_DETAIL,
    { type: AMENITY_REQUEST_TYPE, amenityId: amenityId ?? undefined },
    {
      enabled: amenityId !== null,
      select: selectGeneralInfo,
    },
  );
}
src/modules/amenities/hooks/useAmenitySchedule.ts 
'use client';

import { useAuthQuery } from '@/modules/auth/hooks/useAuthQuery';
import { amenityKeys, AMENITY_REQUEST_TYPE } from '../lib/constants';
import { selectSchedule } from '../lib/selectors';
import { GET_AMENITY_SCHEDULE } from '../graphql/amenity.queries';

export function useAmenitySchedule(amenityId: number | null, enabled: boolean) {
  return useAuthQuery(
    amenityKeys.schedule(amenityId ?? -1),
    GET_AMENITY_SCHEDULE,
    { type: AMENITY_REQUEST_TYPE, amenityId: amenityId ?? undefined },
    {
      enabled: enabled && amenityId !== null,
      select: selectSchedule,
    },
  );
}

Behavior is identical to before — this is purely extracting the two select: callback bodies into named functions so the server side can import them too.

4. Server fetchers — same query, same type, same selector

src/modules/amenities/actions/server-amenities.ts 
import 'server-only';

import { cache } from 'react';
import { serverGqlFetch } from '@/lib/graphql/server-gql-fetcher';
import { RateLimitError } from '@/lib/graphql/gql-errors';
import { GET_AMENITY_DETAIL, GET_AMENITY_SCHEDULE } from '../graphql/amenity.queries';
import { AMENITY_REQUEST_TYPE } from '../lib/constants';
import { selectGeneralInfo, selectSchedule } from '../lib/selectors';
import type {
  GeneralInfoValues,
  GetAmenityGeneralInfoResponse,
  AmenitySchedule,
  GetAmenityScheduleDetailsResponse,
} from '@/types/amenities';

export const getAmenityGeneralInfo = cache(
  async (amenityId: number): Promise<GeneralInfoValues | null> => {
    try {
      const data = await serverGqlFetch<GetAmenityGeneralInfoResponse>(
        GET_AMENITY_DETAIL,
        { type: AMENITY_REQUEST_TYPE, amenityId },
      );
      return selectGeneralInfo(data);
    } catch (error) {
      if (error instanceof RateLimitError) {
        console.warn('[getAmenityGeneralInfo] Rate limited — returning null');
        return null;
      }
      console.error('[getAmenityGeneralInfo]', error);
      return null;
    }
  },
);

export const getAmenitySchedule = cache(
  async (amenityId: number): Promise<AmenitySchedule | null> => {
    try {
      const data = await serverGqlFetch<GetAmenityScheduleDetailsResponse>(
        GET_AMENITY_SCHEDULE,
        { type: AMENITY_REQUEST_TYPE, amenityId },
      );
      return selectSchedule(data);
    } catch (error) {
      if (error instanceof RateLimitError) {
        console.warn('[getAmenitySchedule] Rate limited — returning null');
        return null;
      }
      console.error('[getAmenitySchedule]', error);
      return null;
    }
  },
);

Zero new GraphQL documents, zero new response types. The only "new" code here is the cache() wrapper and the try/catch — the query, the variables shape, and the data mapping are all the exact same ones the client hooks use.

why the type-check happens in the page, not the fetcher

selectGeneralInfo only returns amenityTypeId, amenityName, amenityDescription — it never selected amenityTypeName, matching what the client hook already does. So "is this Scheduled Admission?" is answered by matching generalInfo.amenityTypeId against the already-fetched amenity types list, rather than adding a new field the client side doesn't use.

5. view-amenity page.tsx — using the reused fetchers

src/app/(protected)/location/[slug]/[fn]/(amenity)/view-amenity/[amenityId]/page.tsx 
import { notFound } from 'next/navigation';
import { getLocationContext } from '@/modules/locations/lib/get-location-context';
import { getLocationFunctions } from '@/modules/locations/actions/server-location-functions';
import { getAmenityTypes } from '@/modules/amenities/actions/server-amenities';
import { getAmenityGeneralInfo, getAmenitySchedule } from '@/modules/amenities/actions/server-amenities';
import { PROVIDE_AMENITIES_SLUG, SCHEDULED_ADMISSION_TYPE_NAME } from '@/modules/amenities/lib/constants';
import { EditAmenity } from '@/modules/amenities/components/AddAmenity/EditAmenity';

interface Props {
  params: Promise<{ slug: string; fn: string; amenityId: number }>;
}

export default async function ViewAmenityPage({ params }: Props) {
  const { slug, fn, amenityId } = await params;
  if (fn !== 'provide-amenities') notFound();

  const found = await getLocationContext(slug);
  if (!found) notFound();
  const locationId = Number(found.node.id);

  // None of these three depend on each other — run together.
  const [generalInfo, amenityTypesResult, functions] = await Promise.all([
    getAmenityGeneralInfo(amenityId),
    getAmenityTypes(),
    getLocationFunctions(String(locationId)),
  ]);

  if (!generalInfo) notFound();

  const amenityTypeName = amenityTypesResult.amenityTypes.find(
    (t) => t.amenityTypeId === generalInfo.amenityTypeId,
  )?.amenityTypeName;

  // Only known after generalInfo + amenityTypes resolve — this fetch
  // can't start any earlier than this.
  const schedule =
    amenityTypeName === SCHEDULED_ADMISSION_TYPE_NAME
      ? await getAmenitySchedule(amenityId)
      : null;

  const amenitiesFn = functions.find((f) => f.slug === PROVIDE_AMENITIES_SLUG);

  return (
    <EditAmenity
      amenityId={amenityId}
      locationId={locationId}
      locationSlug={slug}
      amenityTypes={amenityTypesResult.amenityTypes}
      locationFunctionId={amenitiesFn?.locationFunctionId ?? 0}
      generalInfo={generalInfo}
      schedule={schedule}
    />
  );
}
Net result: GET_AMENITY_DETAIL, GET_AMENITY_SCHEDULE, GetAmenityGeneralInfoResponse, GetAmenityScheduleDetailsResponse — each defined exactly once, imported by both server and client. Mapping logic exists exactly once per query, in selectors.ts, imported by both.
