1. ID vs. slug for the amenity route

decision

Use the amenity ID in the URL. Do not slugify the amenity name.

Locations / FunctionsAmenities
Curated taxonomy — admin-defined, stable, hierarchical. Built for human-readable, shareable URLs. You already have slug-resolution infra (getLocationsTree) for this. User-generated records — created/renamed/deleted freely, no uniqueness guarantee on name. Slugging invites collisions and broken links on rename.

This is the same pattern as /orders/8841 or /issues/492 — opaque, stable identifiers for mutable records. Zero extra infrastructure, never breaks.

Future option if you want it more readable without giving up stability: a slug-id hybrid like /view-amenity/8841-cafe-wifi, where only the leading id resolves the record and the trailing text is cosmetic. Not needed now — flagging for later.

2. Final folder structure

location/[slug]/[fn]/ ├── page.tsx (dispatch — unchanged) ├── render-function-page.tsx (top-level fn switch — unchanged) ├── (amenity)/ route group — adds no URL segment │ ├── layout.tsx NEW — breadcrumb prefix, add/view only │ ├── add-amenity/ │ │ └── page.tsx │ └── view-amenity/ │ └── [amenityId]/ │ └── page.tsx

(amenity) — parentheses mean Next.js treats it as an organizational route group: it groups files in the tree and lets you attach a shared layout, but is stripped from the actual URL. The listing page at [fn]/page.tsx sits outside the group, so the group's breadcrumb layout doesn't apply to it — correct, since the list page shouldn't show "Provide Amenities ›" pointing at itself.

3. Code

render-function-page.tsx

src/app/(protected)/location/[slug]/[fn]/render-function-page.tsx 
import { notFound } from 'next/navigation';
import { getLocationPlans } from '@/modules/plans/actions/server-location-plans';
import { PlanShell } from '@/modules/plans/components/planShell';
import { ConfigurationsShell } from '@/modules/configurations/components/configShell';
import { AmenitiesPage } from '@/modules/amenities/components/AmenitiesPage';

interface Props {
  fn: string;
  locationId: number;
  locationSlug: string;
}

export async function renderFunctionPage({
  fn,
  locationId,
  locationSlug,
}: Props) {
  switch (fn) {
    case 'plan-selection': {
      const locationPlansResult = await getLocationPlans(locationId);
      return (
        <PlanShell data={locationPlansResult} locationId={locationId} />
      );
    }

    case 'configurations': {
      return <ConfigurationsShell locationId={locationId} />;
    }

    case 'provide-amenities': {
      return (
        <AmenitiesPage locationId={locationId} locationSlug={locationSlug} />
      );
    }

    default:
      notFound();
  }
}

(amenity)/layout.tsx — breadcrumb prefix, scoped to create/view only

src/app/(protected)/location/[slug]/[fn]/(amenity)/layout.tsx 
import { Breadcrumbs } from '@/components/layout/Breadcrumbs';
import { getLocationsTree } from '@/modules/locations/actions/server-locations';
import { getNodeAndPathBySlug } from '@/modules/locations/lib/utils';
import { buildFeatureCrumbs } from '@/lib/breadcrumbs';

interface Props {
  children: React.ReactNode;
  params: Promise<{ slug: string; fn: string }>;
}

export default async function AmenityGroupLayout({ children, params }: Props) {
  const { slug, fn } = await params;

  const tree = await getLocationsTree();
  const found = getNodeAndPathBySlug(tree, slug);

  const locationCrumbs =
    found?.path.map((node) => ({
      label: node.name,
      href: `/location/${node.slug}`,
    })) ?? [];

  // "Provide Amenities" crumb only — the leaf (Add/View Amenity) crumb
  // is appended by each page below, since only it knows which leaf it is.
  const featureCrumbs = buildFeatureCrumbs(slug, fn);

  return (
    <div>
      <Breadcrumbs items={[...locationCrumbs, ...featureCrumbs]} />
      {children}
    </div>
  );
}

lib/breadcrumbs.ts

src/lib/breadcrumbs.ts 
export const FN_LABELS: Record<string, string> = {
  'provide-amenities': 'Provide Amenities',
  'plan-selection': 'Plan Selection',
  configurations: 'Configurations',
  tickets: 'Tickets',
};

export function buildFeatureCrumbs(locationSlug: string, fn: string) {
  return [
    {
      label: FN_LABELS[fn] ?? fn,
      href: `/location/${locationSlug}/${fn}`,
    },
  ];
}

add-amenity/page.tsx

src/app/(protected)/location/[slug]/[fn]/(amenity)/add-amenity/page.tsx 
import { notFound } from 'next/navigation';
import { getLocationsTree } from '@/modules/locations/actions/server-locations';
import { getNodeAndPathBySlug } from '@/modules/locations/lib/utils';
import { AddAmenityForm } from '@/modules/amenities/components/forms/AddAmenityForm';

interface Props {
  params: Promise<{ slug: string; fn: string }>;
}

export default async function AddAmenityPage({ params }: Props) {
  const { slug, fn } = await params;
  if (fn !== 'provide-amenities') notFound();

  const tree = await getLocationsTree();
  const found = getNodeAndPathBySlug(tree, slug);
  if (!found) notFound();

  return (
    <AddAmenityForm locationId={Number(found.node.id)} locationSlug={slug} />
  );
}

view-amenity/[amenityId]/page.tsx

src/app/(protected)/location/[slug]/[fn]/(amenity)/view-amenity/[amenityId]/page.tsx 
import { notFound } from 'next/navigation';
import { getLocationsTree } from '@/modules/locations/actions/server-locations';
import { getNodeAndPathBySlug } from '@/modules/locations/lib/utils';
import { getAmenityById } from '@/modules/amenities/actions/server-amenities';
import { ViewAmenityDetail } from '@/modules/amenities/components/ViewAmenityDetail';

interface Props {
  params: Promise<{ slug: string; fn: string; amenityId: string }>;
}

export default async function ViewAmenityPage({ params }: Props) {
  const { slug, fn, amenityId } = await params;
  if (fn !== 'provide-amenities') notFound();

  const tree = await getLocationsTree();
  const found = getNodeAndPathBySlug(tree, slug);
  if (!found) notFound();

  const amenity = await getAmenityById(amenityId);
  if (!amenity) notFound();

  return (
    <ViewAmenityDetail
      amenity={amenity}
      locationId={Number(found.node.id)}
      locationSlug={slug}
    />
  );
}
