1. Cached location lookup — single source, fetched once per request

src/modules/locations/lib/get-location-context.ts 
import { cache } from 'react';
import { getLocationsTree } from '@/modules/locations/actions/server-locations';
import { getNodeAndPathBySlug } from '@/modules/locations/lib/utils';

/**
 * React's cache() memoizes by argument for the lifetime of a single
 * request/render pass. The layout calls this for breadcrumbs/nav, and
 * the leaf page calls it again for locationId — with the same slug,
 * the second call resolves instantly from cache instead of re-fetching
 * and re-walking the tree.
 */
export const getLocationContext = cache(async (slug: string) => {
  const tree = await getLocationsTree();
  return getNodeAndPathBySlug(tree, slug);
});

3. add-amenity/page.tsx — use the cached helper

src/app/(protected)/location/[slug]/[fn]/(amenity)/add-amenity/page.tsx 
import { notFound } from 'next/navigation';
import { getLocationContext } from '@/modules/locations/lib/get-location-context';
import { AddAmenityForm } from '@/modules/amenities/components/forms/AddAmenityForm';

interface Props {
  params: Promise<{ slug: string; fn: string }>;
}

export default async function AddAmenityPage({ params }: Props) {
  const { slug, fn } = await params;
  if (fn !== 'provide-amenities') notFound();

  const found = await getLocationContext(slug);
  if (!found) notFound();

  return (
    <AddAmenityForm locationId={Number(found.node.id)} locationSlug={slug} />
  );
}

4. view-amenity/[amenityId]/page.tsx — cached helper + parallel fetch

src/app/(protected)/location/[slug]/[fn]/(amenity)/view-amenity/[amenityId]/page.tsx 
import { notFound } from 'next/navigation';
import { getLocationContext } from '@/modules/locations/lib/get-location-context';
import { getAmenityById } from '@/modules/amenities/actions/server-amenities';
import { ViewAmenityDetail } from '@/modules/amenities/components/ViewAmenityDetail';

interface Props {
  params: Promise<{ slug: string; fn: string; amenityId: string }>;
}

export default async function ViewAmenityPage({ params }: Props) {
  const { slug, fn, amenityId } = await params;
  if (fn !== 'provide-amenities') notFound();

  // Independent fetches — run together instead of one after another.
  // Total wait becomes max(a, b) instead of a + b.
  const [found, amenity] = await Promise.all([
    getLocationContext(slug),
    getAmenityById(amenityId),
  ]);

  if (!found) notFound();
  if (!amenity) notFound();

  return (
    <ViewAmenityDetail
      amenity={amenity}
      locationId={Number(found.node.id)}
      locationSlug={slug}
    />
  );
}

6. Optional — hover-prefetch for the Add Amenity button

<Link> gets automatic prefetch on viewport-visibility for free. A button doesn't — but you can opt in manually:

src/modules/amenities/hooks/useAmenityNavigation.ts 
'use client';

import { useRouter } from 'next/navigation';

export function useAmenityNavigation(locationSlug: string) {
  const router = useRouter();
  const base = `/location/${locationSlug}/provide-amenities`;

  return {
    goToAmenitiesList: () => router.push(base),
    goToAddAmenity: () => router.push(`${base}/add-amenity`),
    goToViewAmenity: (amenityId: string | number) =>
      router.push(`${base}/view-amenity/${amenityId}`),
    prefetchAddAmenity: () => router.prefetch(`${base}/add-amenity`),
    prefetchViewAmenity: (amenityId: string | number) =>
      router.prefetch(`${base}/view-amenity/${amenityId}`),
  };
}
src/modules/amenities/components/AmenitiesPage.tsx (excerpt) 
const { goToAddAmenity, prefetchAddAmenity } = useAmenityNavigation(locationSlug);

<button
  type="button"
  onClick={goToAddAmenity}
  onMouseEnter={prefetchAddAmenity}
  className="btn-primary"
>
  Add Amenity +
</button>

Prefetch kicks off on hover, so by the time the click registers, the route's RSC payload is often already in flight — same head start a <Link> gets automatically.

net effect

One tree fetch per request instead of two, the two remaining independent fetches run concurrently instead of sequentially, and a route-scoped skeleton replaces the full-pane freeze — the visible "whole page reloading" feeling should be gone or reduced to a brief, expected skeleton flash.
