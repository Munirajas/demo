why this is the right layer

[fn]/layout.tsx physically cannot see add-amenity or view-amenity/222 — Next.js only gives a layout the params for its own segment and everything above it, never below. Rather than threading extra props down through every nested page back up to a layout that can't receive them, we read the route client-side with useSelectedLayoutSegments(), which returns exactly "everything rendered below this layout" live, for any current or future route. One registry entry per new route segment is the only future maintenance this needs.

2. Breadcrumbs.tsx — updated

src/components/layout/Breadcrumbs.tsx 
'use client';

import { useMemo, useState } from 'react';
import Link from 'next/link';
import { usePathname, useSelectedLayoutSegments } from 'next/navigation';
import type { BreadcrumbItem } from '@/types/locations';
import { NESTED_SEGMENT_LABELS } from '@/lib/breadcrumb-segments';

const MAX_LOCATION_CRUMBS = 3;

interface Props {
  items: BreadcrumbItem[]; // location path + fn crumb, from buildBreadcrumbs — unchanged input
}

export function Breadcrumbs({ items }: Props) {
  const [open, setOpen] = useState(false);
  const pathname = usePathname();

  // Everything rendered below the [fn] layout: [] | ['add-amenity'] |
  // ['view-amenity', '222'] | any future nested route, automatically.
  const nestedSegments = useSelectedLayoutSegments();

  const allItems = useMemo(
    () => appendNestedCrumbs(items, nestedSegments, pathname),
    [items, nestedSegments, pathname],
  );

  if (!allItems.length) return null;

  const fnItem = allItems[allItems.length - 1];
  const locationItems = allItems.slice(0, -1);
  const needsCollapse = locationItems.length > MAX_LOCATION_CRUMBS;
  const visibleLocations = needsCollapse
    ? [locationItems[0], ...locationItems.slice(-(MAX_LOCATION_CRUMBS - 1))]
    : locationItems;
  const hiddenLocations = needsCollapse
    ? locationItems.slice(1, -(MAX_LOCATION_CRUMBS - 1))
    : [];

  return (
    <nav className="flex items-center gap-1 flex-wrap text-sm min-w-0">
      {visibleLocations.map((item, i) => (
        <span key={item.href ?? item.label} className="flex items-center gap-1 min-w-0">
          {i > 0 && <Sep />}
          <Crumb item={item} />
          {needsCollapse && i === 0 && (
            <>
              <Sep />
              <div className="relative">
                <button
                  type="button"
                  onClick={() => setOpen((o) => !o)}
                  className="px-2 py-0.5 text-sm text-gray-500 hover:text-gray-900 hover:bg-gray-50 rounded-md"
                >
                  …
                </button>
                {open && (
                  <>
                    <div className="fixed inset-0 z-40" onClick={() => setOpen(false)} />
                    <div className="absolute left-0 top-full mt-1 min-w-[180px] bg-white rounded-lg shadow-lg border border-gray-100 z-50 overflow-hidden py-1">
                      {hiddenLocations.map((h) => (
                        <Link
                          key={h.href}
                          href={h.href ?? '#'}
                          onClick={() => setOpen(false)}
                          className="block px-3 py-2 text-sm text-gray-600 hover:bg-gray-50 hover:text-gray-900 transition-colors truncate"
                        >
                          {h.label}
                        </Link>
                      ))}
                    </div>
                  </>
                )}
              </div>
            </>
          )}
        </span>
      ))}
      <Sep />
      <span
        className="px-4 py-0.5 font-medium text-blue-700 bg-blue-50 rounded-md"
        title={fnItem.label}
      >
        {fnItem.label}
      </span>
    </nav>
  );
}

/**
 * Appends breadcrumb entries for any route segments rendered below [fn].
 * - A segment found in NESTED_SEGMENT_LABELS starts a new crumb.
 * - A segment NOT found (a dynamic id, e.g. amenityId) is folded into the
 *   previous crumb's href instead of becoming its own entry — so
 *   "view-amenity/222" renders as ONE "View Amenity" link, not two.
 * - If there's nothing nested, the original items (location + fn) pass
 *   through unchanged — this is a pure no-op for existing routes.
 */
function appendNestedCrumbs(
  baseItems: BreadcrumbItem[],
  segments: string[],
  pathname: string,
): BreadcrumbItem[] {
  if (!segments.length || !baseItems.length) return baseItems;

  const suffix = `/${segments.join('/')}`;
  const baseHref = pathname.endsWith(suffix)
    ? pathname.slice(0, -suffix.length) || '/'
    : pathname;

  // The fn crumb was previously always "last" (rendered as the non-link
  // current-page badge). Now that something is nested below it, make sure
  // it's clickable — it slides into the regular Crumb group below.
  const items = [...baseItems];
  items[items.length - 1] = { ...items[items.length - 1], href: baseHref };

  const extra: BreadcrumbItem[] = [];
  let hrefSoFar = baseHref;

  for (const seg of segments) {
    hrefSoFar = `${hrefSoFar}/${seg}`;
    const label = NESTED_SEGMENT_LABELS[seg];
    if (label) {
      extra.push({ label, href: hrefSoFar });
    } else if (extra.length) {
      extra[extra.length - 1] = { ...extra[extra.length - 1], href: hrefSoFar };
    }
  }

  return [...items, ...extra];
}

function Sep() {
  return <span className="text-gray-300 select-none">/</span>;
}

function Crumb({ item }: { item: BreadcrumbItem }) {
  if (item.href) {
    return (
      <Link
        href={item.href}
        className="px-4 py-0.5 text-sm text-left rounded-md transition-colors text-gray-900 hover:text-blue-700 hover:bg-blue-50 font-medium border-transparent cursor-pointer"
        title={item.label}
      >
        {item.label}
      </Link>
    );
  }
  return (
    <span className="px-4 py-0.5 text-sm text-gray-500" title={item.label}>
      {item.label}
    </span>
  );
}

Only additions: the useSelectedLayoutSegments import/call and the appendNestedCrumbs function. The render logic below (collapse, Crumb, final badge) is your existing code, untouched — it already worked generically on whatever array it's handed.

3. Nested segment label registry

src/lib/breadcrumb-segments.ts 
/**
 * Display label for any route segment nested below a location's [fn].
 * Add one line here whenever you add a new nested route (add-amenity,
 * view-amenity, edit-amenity, whatever comes next) — Breadcrumbs.tsx
 * picks it up automatically, no other file needs to change.
 *
 * Segments NOT listed here (dynamic ids like an amenityId) are treated
 * as belonging to the previous crumb rather than shown on their own.
 */
export const NESTED_SEGMENT_LABELS: Record<string, string> = {
  'add-amenity': 'Add Amenity',
  'view-amenity': 'View Amenity',
};

4. Add Amenity — button handler

src/modules/amenities/hooks/useAmenityNavigation.ts 
'use client';

import { useRouter } from 'next/navigation';

/**
 * Central place for amenity route paths. If the URL shape ever changes,
 * it changes once here instead of in every component that links to it.
 */
export function useAmenityNavigation(locationSlug: string) {
  const router = useRouter();
  const base = `/location/${locationSlug}/provide-amenities`;

  return {
    goToAmenitiesList: () => router.push(base),
    goToAddAmenity: () => router.push(`${base}/add-amenity`),
    goToViewAmenity: (amenityId: string | number) =>
      router.push(`${base}/view-amenity/${amenityId}`),
  };
}
src/modules/amenities/components/AmenitiesPage.tsx (excerpt) 
'use client';

import { useAmenityNavigation } from '../hooks/useAmenityNavigation';

export function AmenitiesToolbar({ locationSlug }: { locationSlug: string }) {

  return (
    <button type="button" onClick={goToAddAmenity} className="btn-primary">

    </button>
